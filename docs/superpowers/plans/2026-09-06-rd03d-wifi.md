# RD-03D WiFi Streaming Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The XIAO ESP32-C6 joins home WiFi, advertises itself as `rd03d.local`, and streams the RD-03D's raw UART bytes over TCP port 3333 to a new network mode in the Processing RD03D_RadarChart sketch.

**Architecture:** Two new firmware modules — `wifi_link` (station + reconnect + mDNS) and `stream_server` (single-client TCP listener with non-blocking-ish send that drops a stalled client rather than stalling the radar loop). `main.c` sends the radar's multi-target command at boot and forwards every UART chunk verbatim to the stream server before parsing locally. The sketch gains a `useNetwork` switch that feeds the existing `inputBuffer`/`processInputBuffer` path from a `processing.net.Client`.

**Tech Stack:** ESP-IDF v5.5 (`~/esp/esp-idf-v5.5`), esp_wifi/esp_netif/NVS, `espressif/mdns` managed component, lwIP BSD sockets, Processing `processing.net`.

**Spec:** `docs/superpowers/specs/2026-09-06-rd03d-wifi-design.md`

**Environment (every build shell):** `source ~/esp/esp-idf-v5.5/export.sh` (each Bash call is a fresh shell — combine with idf.py commands on one line). Serial port: `/dev/cu.usbmodem11101` (check `ls /dev/cu.usbmodem*` if missing). Existing firmware and parser are done and hardware-verified; `main/rd03d.c/.h` must not change.

**Credentials note:** WiFi SSID/password live only in the untracked, gitignored `rd03d_uart/sdkconfig` (set in Task 2). They must NEVER appear in committed files, the plan, or commit messages.

- [ ] **Step 3: Create wifi_link.c**

`rd03d_uart/main/wifi_link.c`:

```c
#include "wifi_link.h"

#include <string.h>

#include "esp_event.h"
#include "esp_log.h"
#include "esp_netif.h"
#include "esp_wifi.h"
#include "nvs_flash.h"
#include "sdkconfig.h"

static const char *TAG = "wifi_link";

static void on_wifi_event(void *arg, esp_event_base_t base, int32_t id,
                          void *data)
{
    if (base == WIFI_EVENT && id == WIFI_EVENT_STA_START) {
        esp_wifi_connect();
    } else if (base == WIFI_EVENT && id == WIFI_EVENT_STA_DISCONNECTED) {
        ESP_LOGW(TAG, "disconnected, retrying");
        esp_wifi_connect();
    } else if (base == IP_EVENT && id == IP_EVENT_STA_GOT_IP) {
        ip_event_got_ip_t *e = (ip_event_got_ip_t *)data;
        ESP_LOGI(TAG, "got ip " IPSTR, IP2STR(&e->ip_info.ip));
    }
}

void wifi_link_start(void)
{
    esp_err_t err = nvs_flash_init();
    if (err == ESP_ERR_NVS_NO_FREE_PAGES || err == ESP_ERR_NVS_NEW_VERSION_FOUND) {
        ESP_ERROR_CHECK(nvs_flash_erase());
        err = nvs_flash_init();
    }
    ESP_ERROR_CHECK(err);

    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());
    esp_netif_create_default_wifi_sta();

    wifi_init_config_t init_cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&init_cfg));

    ESP_ERROR_CHECK(esp_event_handler_register(WIFI_EVENT, ESP_EVENT_ANY_ID,
                                               on_wifi_event, NULL));
    ESP_ERROR_CHECK(esp_event_handler_register(IP_EVENT, IP_EVENT_STA_GOT_IP,
                                               on_wifi_event, NULL));

    wifi_config_t sta_cfg = { 0 };
    strlcpy((char *)sta_cfg.sta.ssid, CONFIG_RD03D_WIFI_SSID,
            sizeof(sta_cfg.sta.ssid));
    strlcpy((char *)sta_cfg.sta.password, CONFIG_RD03D_WIFI_PASSWORD,
            sizeof(sta_cfg.sta.password));
    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_STA));
    ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_STA, &sta_cfg));
    ESP_ERROR_CHECK(esp_wifi_start());

    ESP_LOGI(TAG, "wifi starting, ssid=%s", CONFIG_RD03D_WIFI_SSID);
}
```

```c
#include <stddef.h>
#include <stdint.h>

/* Starts a TCP listener on the given port. One client at a time; a new
 * connection replaces the current one. */
void stream_server_start(uint16_t port);

/* Forwards bytes to the connected client, if any (no-op otherwise). Bounded
 * blocking: a stalled client is dropped after a short send timeout rather
 * than stalling the caller. */
void stream_server_send(const uint8_t *data, size_t len);
```

- [ ] **Step 2: Create stream_server.c**

`rd03d_uart/main/stream_server.c`:

```c
#include "stream_server.h"

#include <string.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <netinet/tcp.h>
#include <unistd.h>

#include "freertos/FreeRTOS.h"
#include "freertos/semphr.h"
#include "freertos/task.h"
#include "esp_log.h"

static const char *TAG = "stream_server";

static SemaphoreHandle_t s_lock;
static int s_client_fd = -1;
static uint16_t s_port;

/* Caller must hold s_lock. */
static void close_client_locked(void)
{
    if (s_client_fd >= 0) {
        close(s_client_fd);
        s_client_fd = -1;
        ESP_LOGI(TAG, "client disconnected");
    }
}

static void accept_task(void *arg)
{
    (void)arg;
    int listen_fd = socket(AF_INET, SOCK_STREAM, IPPROTO_IP);
    if (listen_fd < 0) {
        ESP_LOGE(TAG, "socket() failed, server not running");
        vTaskDelete(NULL);
        return;
    }
    int opt = 1;
    setsockopt(listen_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    struct sockaddr_in addr = {
        .sin_family = AF_INET,
        .sin_port = htons(s_port),
        .sin_addr.s_addr = htonl(INADDR_ANY),
    };
    if (bind(listen_fd, (struct sockaddr *)&addr, sizeof(addr)) < 0 ||
        listen(listen_fd, 1) < 0) {
        ESP_LOGE(TAG, "bind/listen failed, server not running");
        close(listen_fd);
        vTaskDelete(NULL);
        return;
    }
    ESP_LOGI(TAG, "listening on port %u", (unsigned)s_port);

    for (;;) {
        int fd = accept(listen_fd, NULL, NULL);
        if (fd < 0) {
            continue;
        }
        const struct timeval send_timeout = { .tv_sec = 0, .tv_usec = 100000 };
        setsockopt(fd, SOL_SOCKET, SO_SNDTIMEO, &send_timeout,
                   sizeof(send_timeout));
        int nodelay = 1;
        setsockopt(fd, IPPROTO_TCP, TCP_NODELAY, &nodelay, sizeof(nodelay));

        xSemaphoreTake(s_lock, portMAX_DELAY);
        close_client_locked();
        s_client_fd = fd;
        xSemaphoreGive(s_lock);
        ESP_LOGI(TAG, "client connected");
    }
}

void stream_server_start(uint16_t port)
{
    s_port = port;
    s_lock = xSemaphoreCreateMutex();
    xTaskCreate(accept_task, "stream_accept", 4096, NULL, 5, NULL);
}

void stream_server_send(const uint8_t *data, size_t len)
{
    xSemaphoreTake(s_lock, portMAX_DELAY);
    if (s_client_fd >= 0) {
        ssize_t sent = send(s_client_fd, data, len, 0);
        if (sent < 0 || (size_t)sent != len) {
            ESP_LOGW(TAG, "send failed, dropping client");
            close_client_locked();
        }
    }
    xSemaphoreGive(s_lock);
}
```

- [ ] **Step 3: Add the network functions**

Add these two functions after `setup()`:

```java
void connectNetwork() {
  try {
    netClient = new Client(this, networkHost, networkPort);
  } catch (Exception e) {
    println("Network connect failed: " + e.getMessage());
    netClient = null;
  }
  lastConnectAttemptMs = millis();
}

void pollNetwork() {
  if (netClient == null || !netClient.active()) {
    if (millis() - lastConnectAttemptMs > 2000) {
      if (netClient != null) netClient.stop();
      connectNetwork();
    }
    return;
  }
  synchronized(inputBuffer) {
    while (netClient.available() > 0) {
      int nextByte = netClient.read();
      if (nextByte < 0) break;
      inputBuffer.add((byte) nextByte);
    }
    processInputBuffer();
  }
}
```

```markdown
### Serial mode (direct USB bridge)

1. Set `useNetwork = false` at the top of the sketch.
2. Set `serialPortName` if you do not want the first available port, and
   `baudRate` to the RD-03D UART baud (256000).
3. Run the sketch.
```
