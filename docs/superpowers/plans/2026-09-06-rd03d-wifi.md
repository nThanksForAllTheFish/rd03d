The Processing sketch is untouched (keeps its USB serial mode).

**Tech Stack:** ESP-IDF v5.5 (`~/esp/esp-idf-v5.5`), esp_wifi/esp_netif/NVS, `espressif/mdns` managed component, esp_http_server with WebSocket support, vanilla HTML/JS canvas.

**Spec:** `docs/superpowers/specs/2026-09-06-rd03d-wifi-design.md` (revised: web UI replaced the raw-TCP + Processing network mode after Task 2 landed)

**Environment (every build shell):** `source ~/esp/esp-idf-v5.5/export.sh` (each Bash call is a fresh shell — combine with idf.py commands on one line). Serial port: `/dev/cu.usbmodem11101` (check `ls /dev/cu.usbmodem*` if missing). `main/rd03d.c/.h` must not change. Device IP on this network: 192.168.1.186 (DHCP; prefer `rd03d.local` once Task 3 lands).

### Task 4: HTTP + WebSocket server pushing parsed JSON

> **As-built deviation (recorded during execution):** the prescribed `ws_get`
> handshake registration (`req->method == HTTP_GET` populating `s_ws_fds`) never
> fired on ESP-IDF v5.5 — the server completes the WS handshake without calling
> the URI handler, so no client was ever registered and no frames were pushed.
> The committed `web_server.c` instead enumerates live sockets on each send via
> `httpd_get_client_list()` + `httpd_ws_get_fd_info()`, pushing to every fd in
> `HTTPD_WS_CLIENT_WEBSOCKET` state and closing any whose send fails. The
> "ws client connected" log line therefore does not exist; WS connectivity is
> verified by receiving frames. Public API and JSON format unchanged.

- [x] **Step 4: Create web_server.c**

`rd03d_uart/main/web_server.c`:

```c
#include "web_server.h"

#include <inttypes.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#include "esp_http_server.h"
#include "esp_log.h"

static const char *TAG = "web_server";

extern const uint8_t index_html_start[] asm("_binary_index_html_start");
extern const uint8_t index_html_end[] asm("_binary_index_html_end");

#define MAX_WS_CLIENTS 4

static httpd_handle_t s_server;
static int s_ws_fds[MAX_WS_CLIENTS]; /* -1 = free; touched only on httpd task */

typedef struct {
    char json[192];
} ws_msg_t;

static esp_err_t root_get(httpd_req_t *req)
{
    httpd_resp_set_type(req, "text/html");
    return httpd_resp_send(req, (const char *)index_html_start,
                           index_html_end - index_html_start);
}

static esp_err_t ws_get(httpd_req_t *req)
{
    if (req->method == HTTP_GET) {
        /* Handshake completed by the server; remember the client. */
        int fd = httpd_req_to_sockfd(req);
        for (int i = 0; i < MAX_WS_CLIENTS; i++) {
            if (s_ws_fds[i] == fd) {
                return ESP_OK;
            }
        }
        for (int i = 0; i < MAX_WS_CLIENTS; i++) {
            if (s_ws_fds[i] < 0) {
                s_ws_fds[i] = fd;
                ESP_LOGI(TAG, "ws client connected (fd %d)", fd);
                return ESP_OK;
            }
        }
        ESP_LOGW(TAG, "ws client limit reached, rejecting fd %d", fd);
        return ESP_FAIL;
    }

    /* We are push-only: drain and ignore anything the client sends. */
    httpd_ws_frame_t frame = { 0 };
    esp_err_t ret = httpd_ws_recv_frame(req, &frame, 0);
    if (ret != ESP_OK) {
        return ret;
    }
    if (frame.len > 0) {
        uint8_t *buf = malloc(frame.len);
        if (buf == NULL) {
            return ESP_ERR_NO_MEM;
        }
        frame.payload = buf;
        ret = httpd_ws_recv_frame(req, &frame, frame.len);
        free(buf);
    }
    return ret;
}

/* Runs on the httpd task via httpd_queue_work. */
static void ws_send_work(void *arg)
{
    ws_msg_t *msg = arg;
    httpd_ws_frame_t frame = {
        .final = true,
        .type = HTTPD_WS_TYPE_TEXT,
        .payload = (uint8_t *)msg->json,
        .len = strlen(msg->json),
    };
    for (int i = 0; i < MAX_WS_CLIENTS; i++) {
        int fd = s_ws_fds[i];
        if (fd < 0) {
            continue;
        }
        if (httpd_ws_get_fd_info(s_server, fd) != HTTPD_WS_CLIENT_WEBSOCKET) {
            s_ws_fds[i] = -1; /* client went away; reap the slot */
            continue;
        }
        if (httpd_ws_send_frame_async(s_server, fd, &frame) != ESP_OK) {
            ESP_LOGW(TAG, "ws send failed, dropping fd %d", fd);
            httpd_sess_trigger_close(s_server, fd);
            s_ws_fds[i] = -1;
        }
    }
    free(msg);
}

void web_server_start(void)
{
    for (int i = 0; i < MAX_WS_CLIENTS; i++) {
        s_ws_fds[i] = -1;
    }

    httpd_config_t cfg = HTTPD_DEFAULT_CONFIG();
    if (httpd_start(&s_server, &cfg) != ESP_OK) {
        ESP_LOGE(TAG, "httpd_start failed, web UI unavailable");
        s_server = NULL;
        return;
    }

    static const httpd_uri_t root_uri = {
        .uri = "/",
        .method = HTTP_GET,
        .handler = root_get,
    };
    static const httpd_uri_t ws_uri = {
        .uri = "/ws",
        .method = HTTP_GET,
        .handler = ws_get,
        .is_websocket = true,
    };
    ESP_ERROR_CHECK(httpd_register_uri_handler(s_server, &root_uri));
    ESP_ERROR_CHECK(httpd_register_uri_handler(s_server, &ws_uri));
    ESP_LOGI(TAG, "http server listening on port %d", cfg.server_port);
}

void web_server_send_frame(const rd03d_frame_t *f, uint32_t dropped,
                           uint32_t bad)
{
    if (s_server == NULL) {
        return;
    }
    ws_msg_t *msg = malloc(sizeof(*msg));
    if (msg == NULL) {
        return;
    }

    int off = snprintf(msg->json, sizeof(msg->json), "{\"t\":[");
    for (int i = 0; i < RD03D_NUM_TARGETS; i++) {
        if (off <= 0 || off >= (int)sizeof(msg->json)) {
            break;
        }
        const rd03d_target_t *t = &f->targets[i];
        const char *sep = (i < RD03D_NUM_TARGETS - 1) ? "," : "";
        if (t->present) {
            off += snprintf(msg->json + off, sizeof(msg->json) - (size_t)off,
                            "{\"x\":%d,\"y\":%d,\"v\":%d}%s",
                            t->x_mm, t->y_mm, t->speed_cms, sep);
        } else {
            off += snprintf(msg->json + off, sizeof(msg->json) - (size_t)off,
                            "null%s", sep);
        }
    }
    if (off > 0 && off < (int)sizeof(msg->json)) {
        off += snprintf(msg->json + off, sizeof(msg->json) - (size_t)off,
                        "],\"dropped\":%" PRIu32 ",\"bad\":%" PRIu32 "}",
                        dropped, bad);
    }
    if (off <= 0 || off >= (int)sizeof(msg->json)) {
        free(msg); /* formatting error or truncation: drop this frame */
        return;
    }

    if (httpd_queue_work(s_server, ws_send_work, msg) != ESP_OK) {
        free(msg);
    }
}
```

```html
<!DOCTYPE html>
<html>

<script>
"use strict";
const RANGE_X_MM = 3000;  // +/-3 m across
const RANGE_Y_MM = 6000;  // 6 m forward
const TRAIL = 120;        // trail points per target
const COLORS = ["#00b4ff", "#00dc78", "#ffb428"];

const canvas = document.getElementById("chart");
const ctx = canvas.getContext("2d");
const statusEl = document.getElementById("status");
const targetsEl = document.getElementById("targets");
const statsEl = document.getElementById("stats");

let frames = 0;
let latest = [null, null, null];
const trails = [[], [], []];

function resize() {
  const dpr = window.devicePixelRatio || 1;
  canvas.width = Math.round(canvas.clientWidth * dpr);
  canvas.height = Math.round(canvas.clientHeight * dpr);
}
window.addEventListener("resize", resize);

// Uniform mm->px scale so range arcs stay circular.
function scale() {
  return Math.min(canvas.width / (2 * RANGE_X_MM), canvas.height / RANGE_Y_MM);
}

function toPx(xMm, yMm) {
  const s = scale();
  return [canvas.width / 2 + xMm * s, canvas.height - yMm * s];
}

function drawGrid() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  const dpr = window.devicePixelRatio || 1;
  const [ox, oy] = toPx(0, 0);
  const s = scale();

  ctx.lineWidth = 1;
  ctx.strokeStyle = "#2c2c2c";
  ctx.fillStyle = "#666";
  ctx.font = (12 * dpr) + "px system-ui";

  for (let r = 1000; r <= RANGE_Y_MM; r += 1000) {
    ctx.beginPath();
    ctx.arc(ox, oy, r * s, Math.PI, 2 * Math.PI);
    ctx.stroke();
    ctx.fillText((r / 1000) + " m", ox + 6 * dpr, oy - r * s - 4 * dpr);
  }

  ctx.strokeStyle = "#3a3a3a";
  ctx.beginPath();
  ctx.moveTo(ox, oy); ctx.lineTo(ox, oy - RANGE_Y_MM * s);
  ctx.moveTo(ox - RANGE_X_MM * s, oy); ctx.lineTo(ox + RANGE_X_MM * s, oy);
  ctx.stroke();
}

function render() {
  drawGrid();
  const dpr = window.devicePixelRatio || 1;
  for (let i = 0; i < 3; i++) {
    ctx.fillStyle = COLORS[i];
    const tr = trails[i];
    for (let j = 0; j < tr.length; j++) {
      const [px, py] = toPx(tr[j].x, tr[j].y);
      ctx.globalAlpha = 0.1 + 0.3 * ((j + 1) / tr.length);
      ctx.beginPath();
      ctx.arc(px, py, 2.5 * dpr, 0, 7);
      ctx.fill();
    }
    ctx.globalAlpha = 1;
    if (latest[i]) {
      const [px, py] = toPx(latest[i].x, latest[i].y);
      ctx.beginPath();
      ctx.arc(px, py, 6 * dpr, 0, 7);
      ctx.fill();
    }
  }
  requestAnimationFrame(render);
}

function onMessage(ev) {
  let msg;
  try { msg = JSON.parse(ev.data); } catch (e) { return; }
  frames++;
  latest = msg.t;
  for (let i = 0; i < 3; i++) {
    if (msg.t[i]) {
      trails[i].push({ x: msg.t[i].x, y: msg.t[i].y });
      if (trails[i].length > TRAIL) trails[i].shift();
    }
  }
  statsEl.textContent =
    "frames " + frames + " · dropped " + msg.dropped + " · bad " + msg.bad;
  targetsEl.innerHTML = msg.t.map(function (t, i) {
    return t
      ? '<span style="color:' + COLORS[i] + '">T' + (i + 1) +
        " x=" + t.x + " y=" + t.y + " v=" + t.v + "</span>"
      : '<span style="color:#555">T' + (i + 1) + " —</span>";
  }).join(" ");
}

function connect() {
  const ws = new WebSocket("ws://" + location.host + "/ws");
  ws.onopen = function () {
    statusEl.textContent = "connected";
    statusEl.className = "ok";
  };
  ws.onclose = function () {
    statusEl.textContent = "disconnected — retrying";
    statusEl.className = "bad";
    setTimeout(connect, 2000);
  };
  ws.onerror = function () { ws.close(); };
  ws.onmessage = onMessage;
}

resize();
connect();
render();
</script>
</body>
</html>
```

```markdown
`sdkconfig`, 2.4 GHz WPA2 networks only) and serves a live radar chart:

- Open **http://rd03d.local** from any browser on the same network (Mac or
  phone). The page shows targets on an XY chart with trails, plus link stats.
- Data path: `GET /ws` WebSocket pushes one JSON message per radar frame,
  e.g. `{"t":[{"x":-551,"y":550,"v":0},null,null],"dropped":0,"bad":0}`
  (x/y in mm, v in cm/s, absent targets `null`).
- At boot the firmware puts the radar in multi-target mode itself; viewers
  never send radar commands.
- Untethered use: power the XIAO from a USB power bank; the USB console is
  optional diagnostics only.
```
