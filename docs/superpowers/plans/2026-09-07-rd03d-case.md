`case/fusion_scripts/run_fusion.py`:

```python
#!/usr/bin/env python3
"""Send a Fusion API script to the local Fusion MCP server, or grab a screenshot.

Usage:
  run_fusion.py script <file.py> [--read-only]
  run_fusion.py screenshot <out.png> [direction]
"""
import base64
import json
import sys
import urllib.request

URL = "http://127.0.0.1:27182/mcp"


def rpc(session, payload):
    headers = {"Content-Type": "application/json",
               "Accept": "application/json, text/event-stream"}
    if session:
        headers["MCP-Session-Id"] = session
    req = urllib.request.Request(URL, data=json.dumps(payload).encode(),
                                 headers=headers)
    with urllib.request.urlopen(req, timeout=300) as r:
        sid = r.headers.get("MCP-Session-Id", session)
        body = r.read().decode()
    return sid, (json.loads(body) if body.strip() else None)


def connect():
    sid, _ = rpc(None, {"jsonrpc": "2.0", "id": 1, "method": "initialize",
                        "params": {"protocolVersion": "2025-03-26",
                                   "capabilities": {},
                                   "clientInfo": {"name": "rd03d-case-builder",
                                                  "version": "1.0"}}})
    rpc(sid, {"jsonrpc": "2.0", "method": "notifications/initialized"})
    return sid


def call(sid, name, args):
    _, resp = rpc(sid, {"jsonrpc": "2.0", "id": 2, "method": "tools/call",
                        "params": {"name": name, "arguments": args}})
    if resp is None:
        raise SystemExit("empty MCP response")
    if "error" in resp:
        raise SystemExit(f"MCP error: {resp['error']}")
    return resp["result"].get("content", [])


def main():
    if len(sys.argv) < 3:
        raise SystemExit(__doc__)
    mode = sys.argv[1]
    sid = connect()

    if mode == "script":
        src = open(sys.argv[2]).read()
        obj = {"script": src}
        if "--read-only" in sys.argv:
            obj["readOnly"] = True
        content = call(sid, "fusion_mcp_execute",
                       {"featureType": "script", "object": obj})
        ok = True
        for c in content:
            if c.get("type") != "text":
                continue
            try:
                d = json.loads(c["text"])
                print(d.get("message", c["text"]))
                if d.get("success") is False:
                    ok = False
            except json.JSONDecodeError:
                print(c["text"])
        sys.exit(0 if ok else 1)

    if mode == "screenshot":
        out = sys.argv[2]
        args = {"queryType": "screenshot", "width": 1200, "height": 900}
        if len(sys.argv) > 3 and not sys.argv[3].startswith("--"):
            args["direction"] = sys.argv[3]
        content = call(sid, "fusion_mcp_read", args)
        for c in content:
            if c.get("type") == "image" and c.get("data"):
                open(out, "wb").write(base64.b64decode(c["data"]))
                print("saved", out)
                return
            if c.get("type") == "text":
                try:
                    d = json.loads(c["text"])
                    img = d.get("imageData") or d.get("data")
                    if img:
                        open(out, "wb").write(base64.b64decode(img))
                        print("saved", out)
                        return
                    print(c["text"][:400])
                except json.JSONDecodeError:
                    print(c["text"][:400])
        raise SystemExit("no image in response")

    raise SystemExit(__doc__)


if __name__ == "__main__":
    main()
```

`case/fusion_scripts/02_backplate.py` — intent (prescriptive) and starting code. The component named `BackPlate` under the root, containing:
1. Plate: rectangle (intW−2·rimGap) × (intH−2·rimGap) centered on origin, extruded z=−backT..0.
2. Tape recess: 34×38 pocket, 0.6 deep, cut into the z=−backT face.
3. Snap grooves: on each ±X edge at y=±12, a 6.2 wide × 0.6 deep groove through the plate thickness (cut from the edge inward).
4. Radar bay on the +Z face: perimeter fence 1.5 thick around (radarW+2·boardClear) × (radarH+2·boardClear) centered at (radarCx, 0), fence from z=0 to z=11; two crossbar supports 4 wide spanning the bay interior at y=±18, z=0..6.65 (board back rests on them → patch face lands at z=13).
5. XIAO bay: four 3×3 posts (z=0..postH) at the corners of (xiaoW−2)×(xiaoH−2) centered at (xiaoCx, 0); perimeter fence 1.5 thick, z=0..5, around (xiaoW+2·boardClear)×(xiaoH+2·boardClear), with the +X side of the fence OMITTED (USB cable path).

```python
import adsk.core
import adsk.fusion


def p(des, name):
    return des.userParameters.itemByName(name).value  # cm


def rect(sk, cx, cy, w, h):
    sk.sketchCurves.sketchLines.addTwoPointRectangle(
        adsk.core.Point3D.create(cx - w / 2, cy - h / 2, 0),
        adsk.core.Point3D.create(cx + w / 2, cy + h / 2, 0))


def extrude(comp, profiles, z0, z1, op):
    ext = comp.features.extrudeFeatures
    inp = ext.createInput(profiles, op)
    start = adsk.fusion.OffsetStartDefinition.create(
        adsk.core.ValueInput.createByReal(z0))
    inp.startExtent = start
    inp.setOneSideExtent(
        adsk.fusion.DistanceExtentDefinition.create(
            adsk.core.ValueInput.createByReal(z1 - z0)),
        adsk.fusion.ExtentDirections.PositiveExtentDirection)
    return ext.add(inp)


def collection(items):
    coll = adsk.core.ObjectCollection.create()
    for i in items:
        coll.add(i)
    return coll


def run(_context: str):
    app = adsk.core.Application.get()
    des = adsk.fusion.Design.cast(app.activeProduct)
    root = des.rootComponent

    for occ in list(root.occurrences):
        if occ.component.name == "BackPlate":
            occ.deleteMe()
            print("removed existing BackPlate")

    occ = root.occurrences.addNewComponent(adsk.core.Matrix3D.create())
    comp = occ.component
    comp.name = "BackPlate"

    intW, intH = p(des, "intW"), p(des, "intH")
    backT, rimGap = p(des, "backT"), p(des, "rimGap")
    tape = p(des, "tapeRecess")
    rW, rH, rT = p(des, "radarW"), p(des, "radarH"), p(des, "radarT")
    rCx = p(des, "radarCx")
    xW, xH, xCx = p(des, "xiaoW"), p(des, "xiaoH"), p(des, "xiaoCx")
    postH, clear = p(des, "postH"), p(des, "boardClear")
    intD, wall, winT = p(des, "intD"), p(des, "wall"), p(des, "windowT")
    MM = 0.1

    # 1. plate
    sk = comp.sketches.add(comp.xYConstructionPlane)
    rect(sk, 0, 0, intW - 2 * rimGap, intH - 2 * rimGap)
    extrude(comp, sk.profiles.item(0), -backT, 0,
            adsk.fusion.FeatureOperations.NewBodyFeatureOperation)
    print("plate ok")

    # 2. tape recess (cut upward from below)
    sk = comp.sketches.add(comp.xYConstructionPlane)
    rect(sk, 0, 0, 34 * MM, 38 * MM)
    extrude(comp, sk.profiles.item(0), -backT, -backT + tape,
            adsk.fusion.FeatureOperations.CutFeatureOperation)
    print("tape recess ok")

    # 3. snap grooves at plate +/-X edges, y=+/-12
    sk = comp.sketches.add(comp.xYConstructionPlane)
    for sx in (-1, 1):
        edge = sx * (intW / 2 - rimGap)
        for sy in (-1, 1):
            rect(sk, edge - sx * 0.03, sy * 12 * MM, 0.12, 6.2 * MM)
    profs = collection([sk.profiles.item(i) for i in range(sk.profiles.count)])
    extrude(comp, profs, -backT, 0,
            adsk.fusion.FeatureOperations.CutFeatureOperation)
    print("snap grooves ok")

    # 4. radar bay: fence ring + crossbars
    bayW, bayH = rW + 2 * clear, rH + 2 * clear
    sk = comp.sketches.add(comp.xYConstructionPlane)
    rect(sk, rCx, 0, bayW + 2 * 1.5 * MM, bayH + 2 * 1.5 * MM)
    rect(sk, rCx, 0, bayW, bayH)
    ring = None
    for i in range(sk.profiles.count):
        pr = sk.profiles.item(i)
        if pr.profileLoops.count == 2:
            ring = pr
    extrude(comp, ring, 0, 11 * MM,
            adsk.fusion.FeatureOperations.JoinFeatureOperation)
    sk = comp.sketches.add(comp.xYConstructionPlane)
    for sy in (-1, 1):
        rect(sk, rCx, sy * 18 * MM, bayW, 4 * MM)
    profs = collection([sk.profiles.item(i) for i in range(sk.profiles.count)])
    board_back_z = intD - 1 * MM - rT  # patch face lands 1mm behind front wall
    extrude(comp, profs, 0, board_back_z,
            adsk.fusion.FeatureOperations.JoinFeatureOperation)
    print("radar bay ok, support top z(cm)=", round(board_back_z, 3))

    # 5. xiao bay: 4 posts + 3-sided fence (open toward +X)
    sk = comp.sketches.add(comp.xYConstructionPlane)
    for sx in (-1, 1):
        for sy in (-1, 1):
            rect(sk, xCx + sx * (xW / 2 - 1.5 * MM),
                 sy * (xH / 2 - 1.5 * MM), 3 * MM, 3 * MM)
    profs = collection([sk.profiles.item(i) for i in range(sk.profiles.count)])
    extrude(comp, profs, 0, postH,
            adsk.fusion.FeatureOperations.JoinFeatureOperation)
    fW, fH = xW + 2 * clear, xH + 2 * clear
    t = 1.5 * MM
    sk = comp.sketches.add(comp.xYConstructionPlane)
    rect(sk, xCx - fW / 2 - t / 2, 0, t, fH + 2 * t)          # -X side
    rect(sk, xCx, fH / 2 + t / 2, fW + 2 * t, t)              # +Y side
    rect(sk, xCx, -(fH / 2 + t / 2), fW + 2 * t, t)           # -Y side
    profs = collection([sk.profiles.item(i) for i in range(sk.profiles.count)])
    extrude(comp, profs, 0, 5 * MM,
            adsk.fusion.FeatureOperations.JoinFeatureOperation)
    print("xiao bay ok")

    print("BackPlate bodies:", comp.bRepBodies.count)
```

### Task 5: Import boards, position, interference check

**Files:**

- [x] **Step 1: Write the script**

Intent (prescriptive): import both STEP files as components into the design, orient/translate them into their bays (radar patch-side facing +Z at z_back=intD−1mm−radarT, XIAO on its posts at z=postH), then run interference analysis between every board and every case component; print each pair's interfering volume. The board STEPs' native orientations are unknown — the script prints each imported component's bounding box; the implementer adjusts the rotation matrix and re-runs until the bbox matches the expected footprint, THEN fine-tunes position. Starting code:

```python
import adsk.core
import adsk.fusion

RADAR = "/Users/me/Downloads/rd-03d-sensor-1/RD-03D.step"
XIAO = ("/Users/me/Downloads/seeed-studio-xiao-esp32-c6-1/"
        "Seeed Studio XIAO ESP32-C6.step")


def p(des, name):
    return des.userParameters.itemByName(name).value


def bbox_of(occ):
    bb = None
    for body in occ.bRepBodies:
        b = body.boundingBox
        if bb is None:
            bb = adsk.core.BoundingBox3D.create(b.minPoint.copy(), b.maxPoint.copy())
        else:
            bb.combine(b)
    return bb


def run(_context: str):
    app = adsk.core.Application.get()
    des = adsk.fusion.Design.cast(app.activeProduct)
    root = des.rootComponent

    for occ in list(root.occurrences):
        if occ.component.name.startswith(("RD-03D", "Seeed", "XIAO")):
            occ.deleteMe()
            print("removed prior import:", occ.name)

    imp = app.importManager
    before = root.occurrences.count
    imp.importToTarget(imp.createSTEPImportOptions(RADAR), root)
    radar_occ = root.occurrences.item(before)
    imp.importToTarget(imp.createSTEPImportOptions(XIAO), root)
    xiao_occ = root.occurrences.item(before + 1)

    for name, occ in (("radar", radar_occ), ("xiao", xiao_occ)):
        bb = bbox_of(occ)
        print(name, "bbox(cm): x %.2f..%.2f  y %.2f..%.2f  z %.2f..%.2f" %
              (bb.minPoint.x, bb.maxPoint.x, bb.minPoint.y, bb.maxPoint.y,
               bb.minPoint.z, bb.maxPoint.z))

    # TRANSFORMS: start with identity + translation to bay centers; the
    # implementer sets the rotation after reading the printed bboxes.
    # Target (cm): radar footprint 1.51 x 4.40, patch face at z = intD-0.1;
    #              xiao footprint 2.25 x 1.78 sitting at z = postH.
    intD, rT = p(des, "intD"), p(des, "radarT")
    rCx, xCx, postH = p(des, "radarCx"), p(des, "xiaoCx"), p(des, "postH")

    t = adsk.core.Matrix3D.create()
    t.translation = adsk.core.Vector3D.create(rCx, 0, intD - 0.1 - rT)
    radar_occ.transform2 = t
    t = adsk.core.Matrix3D.create()
    t.translation = adsk.core.Vector3D.create(xCx, 0, postH)
    xiao_occ.transform2 = t
    print("initial placement done - check bboxes/screenshots and refine")

    # Interference: boards vs case components
    ents = adsk.core.ObjectCollection.create()
    for occ in root.occurrences:
        for b in occ.bRepBodies:
            ents.add(b)
    inp = des.createInterferenceInput(ents)
    res = des.analyzeInterference(inp)
    if res.count == 0:
        print("INTERFERENCE: none")
    for i in range(res.count):
        r = res.item(i)
        print("INTERFERENCE: %.4f cm^3 between %s and %s" %
              (r.interferenceVolume, r.entityOne.parentComponent.name,
               r.entityTwo.parentComponent.name))
```

**2026-09-09: radar clips go to THREE with NO relief slots; XIAO clips keep
theirs.** One approved change in one commit; branch `fix/case-clip-revision`.
Supersedes the four-clip arrangement of 2026-09-08.

*Reasoning (the whole point of the change).* The radar lip's flat underside
sits **11.80 mm** above the plate, so the 1.5 mm PETG fence wall under it is
already an 11.8 mm cantilever. Deflecting its tip the 0.60 mm needed to
clear the board edge puts peak surface strain at 3·t·d/(2·L²) =
3·1.5·0.60/(2·11.8²) ≈ **1 %** — a quarter of PETG's ~4–5 % yield — for of
order **12 N** at the lip. A 7 × 1.5 mm band at that length does not need to
be isolated from its neighbours to flex, and the slots were actively harmful:
they cut through the wall whose real job is restraining the radar board
laterally. So all three radar clips now pass an **empty slot list**; the local
wall raise and the lip/ramp are untouched. Dropping the slots also freed the
budget for a third clip, and it un-does the 1.7 × 1.25 × 3 mm nick the old +X
inboard slot took out of the XIAO's +Y fence bar (recorded as cosmetic on
2026-09-08) — that wall is now continuous from the step-4b notch edge
(y = 9.4) all the way to the plate outline.
**The XIAO clips were NOT changed, and the change request's proposal to strip
their slots too was declined on the numbers.** Their lip is only **4.30 mm**
up. Wall stiffness goes as t³/L³, so a slot-free 1.5 mm wall at 4.3 mm is
(1.5/1.0)³ · (11.8/4.3)³ ≈ **22×** stiffer than the thinned, slotted finger —
of order **65 N** for 0.2 mm of travel, i.e. the PCB would bow before the wall
moved. Slots from z = 0, 1.0 mm thinning and 0.34 mm grab all stay as built.
